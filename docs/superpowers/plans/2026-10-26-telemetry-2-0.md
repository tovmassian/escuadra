# Telemetry 2.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship #221's statistics: the 13-event allowlist with the squad id, context on every
signal, sessions that renew on a return from the background, and a persisted offline queue.

**Architecture:** Pure decisions live in `lib/telemetryEvents.ts`, `lib/telemetryQueue.ts` and
`lib/roundStats.ts`, all Vitest-tested. `lib/telemetry.ts` stays the SDK's only importer. It
sends through the SDK's `queue()`/`flush()` with a store adapter over
`stores/progress.ts`'s `telemetryQueue`. Screens pass `from`/`source` as route params and call
`track()`.

**Tech Stack:** Expo SDK 58, React Native 0.88, zustand + AsyncStorage, `@telemetrydeck/sdk`
2.0.4, Vitest (node).

**Spec:** `docs/superpowers/specs/2026-10-26-telemetry-2-0-design.md` (the data dictionary). Issue #221.

## Global Constraints

- Node 24 (`.nvmrc`). Run `npm run check` before claiming done.
- No new dependency, npm script or `.gitignore` line (the fingerprint).
- Nothing in a payload names a player or carries free text. A team appears only as its squad id.
- TypeScript strict with `noUncheckedIndexedAccess`; never loosen it.
- `telemetrySdk.test.ts` and `telemetryIdentifier.test.ts` guard promises. Never edit an
  assertion. Fakes and setup may change where the code under them changed (say so in the PR).
- About's English wording is the owner's (guardrail 7). No `i18n/*.ts` `about` change on this
  branch until the owner agrees. The draft goes in the PR body.
- Tests live in `__tests__/` beside the code. No network, no real clock.
- Screens are not unit-tested. Their logic goes in pure `lib/` helpers.

```ts
export const TELEMETRY_CONTEXT = [
  'TelemetryDeck.AppInfo.version',
  'TelemetryDeck.AppInfo.buildNumber',
  'TelemetryDeck.Device.platform',
  'TelemetryDeck.Device.systemMajorVersion',
  'update',
  'language',
] as const;
export interface TelemetryContextInput {
  os: string;
  appVersion: string | null;
  buildNumber: string | null;
  osVersion: string | null;
  update: string | null;
  language: Language | null;
}
export function contextPayload(c: TelemetryContextInput): Record<string, string> {
  const platform = c.os === 'ios' ? 'iOS' : c.os === 'android' ? 'Android' : null;
  const major = c.osVersion?.split('.')[0] || null;
  const values: Record<TelemetryContextKey, string | null> = {
    'TelemetryDeck.AppInfo.version': c.appVersion,
    'TelemetryDeck.AppInfo.buildNumber': c.buildNumber,
    'TelemetryDeck.Device.platform': platform,
    'TelemetryDeck.Device.systemMajorVersion': major,
    update: c.update,
    language: c.language,
  };
  const out: Record<string, string> = {};
  for (const key of TELEMETRY_CONTEXT) {
    const v = values[key];
    if (v) out[key] = v;
  }
  return out;
}
export const countBucket = (n: number): CountBucket =>
  n <= 0 ? '0' : n === 1 ? '1' : n <= 5 ? '2-5' : '6+';
export const textSizeBucket = (s: number): TextSize =>
  s <= 1.05 ? 'default' : s <= 1.35 ? 'large' : 'largest';
export const roundSeconds = (ms: number): number =>
  Math.min(600, Math.max(0, Math.round(ms / 10_000) * 10));
export type AppPhase = 'foreground' | 'background';
export function nextAppPhase(phase: AppPhase, appState: string) {
  if (appState === 'background') return { phase: 'background' as const, startsSession: false };
  if (appState === 'active')
    return { phase: 'foreground' as const, startsSession: phase === 'background' };
  return { phase, startsSession: false };
}
export function paramOf<T extends string>(value: unknown, allowed: readonly T[]): T | undefined {
  return typeof value === 'string' && (allowed as readonly string[]).includes(value)
    ? (value as T)
    : undefined;
}
export interface LaunchInput {
  languageSource: LanguageSource;
  themePreference: ThemePreference;
  haptics: boolean;
  reduceMotion: boolean | null;
  fontScale: number;
  starredSquads: readonly string[];
  bestScores: Record<string, unknown>;
}
export function launchProps(i: LaunchInput): TelemetryProps['app.launched'] {
  const teams = new Set(Object.keys(i.bestScores).map((key) => key.slice(0, key.lastIndexOf(':'))));
  return {
    languageSource: i.languageSource,
    theme: i.themePreference,
    haptics: i.haptics ? 'on' : 'off',
    ...(i.reduceMotion === null
      ? {}
      : { 'TelemetryDeck.Accessibility.isReduceMotionEnabled': i.reduceMotion }),
    textSize: textSizeBucket(i.fontScale),
    starred: countBucket(i.starredSquads.length),
    teams: countBucket(teams.size),
  };
}
```
