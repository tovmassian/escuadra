# The shot — implementation plan (#129)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** When a question is decided, a ball is struck from the tapped tile: into the escuadra corner on a goal, which then kicks and flashes; wide and off the screen on a miss.

**Architecture:** A pure planner, `lib/shot.ts`, turns the tap point, the corner's position, the verdict and a seed into the ball's position, scale and opacity at equal steps of time, plus the moment the corner reacts. A hook, `components/useShot.ts`, measures the tile, the corner and the overlay, plans the shot and starts one shared clock. A new `ShotOverlay` (the ball and its trail) and the reworked `EscuadraCorner` both read that clock on the UI thread, so the corner turns at the exact frame the ball arrives.

**Tech Stack:** Expo SDK 57, React Native 0.86, Reanimated 4.5 (`useSharedValue`, `useAnimatedStyle`, `withDelay`, `Easing.bezierFn`), Vitest 5 for the pure planner.

**Spec:** [`docs/superpowers/specs/2026-10-15-play-redesign-2-0-design.md`](../specs/2026-10-15-play-redesign-2-0-design.md) §2.4, §6, §8, and issue #129. The shapes come from the design prototype's generator (canvas `Xk2NPWmqxGcMiQqUPUzzav`, `Main.dc.html`). Its fixed 390×844 positions are made relative to the corner, the tap and the screen here.

**Decided:** Depth is **Lob swell** (biggest at the top of the arc, lands at the corner's ball size). The **trail is on**.

## Global Constraints

- `lib/shot.ts` imports no React, no store and does no I/O. It may import `@/theme/tokens` and `@/lib/prng`, as `lib/roundView.ts` does. `sampleShot` carries a `'worklet'` directive so the UI thread can call it. In Vitest the directive is a plain string.
- Tests for `lib/` stay beside the code (`lib/shot.test.ts`), as `lib/` does today.
- Every animation stays at or under 300 ms (`theme/tokens.test.ts` enforces this for `durations`). A goal's beats may chain to about 640 ms. A chain must end inside `durations.autoAdvance` (1,200 ms).
- Colours come from `useThemeColors()` (`success`, `error`, `accentLine`), never a module-scope capture. Sizes and timings come from `theme/tokens.ts`. The shot library's shape numbers (bends, aims, speeds) are data that belong to the planner, and they live in `lib/shot.ts`.
- `spotlightCues.ball` stays 80. `durations.autoAdvance` and "Got it — next" keep their timing. Haptics stay on the tap.
- Reduce Motion (`useReducedMotion`): no flight, no kick, no ring, colour and opacity only. On a goal, the ball fades into the corner.
- TypeScript strict with `noUncheckedIndexedAccess`; never weaken it.
- Run `npm run check` before claiming done (Node 24 is the default `node` here).

// What the screen draws: the plan sampled every millisecond of its flight.
function drawn(plan: ShotPlan): ShotSample[] {
  const out: ShotSample[] = [];
  for (let ms = 0; ms < plan.frames.flight; ms += 1) {
    const at = sampleShot(plan.frames, ms);
    if (at) out.push(at);

  return out;

  const dx = Math.max(box.x - at.x, 0, at.x - (box.x + box.w));
  const dy = Math.max(box.y - at.y, 0, at.y - (box.y + box.h));
  return Math.hypot(dx, dy) - radius(at);
}

import { createRng } from '@/lib/prng';
import { durations, sizes, spotlightMotion } from '@/theme/tokens';

// The shot when a question is decided (#129, spec §6): a ball struck from the
// tapped tile. A goal slows into the escuadra corner through its open,
// lower-left side; a miss keeps its pace, goes wide and leaves the screen,
// never near the corner. Pure: given the tap point, the corner, the verdict
// and a seed, `planShot` returns where the ball is, how big and how opaque, at
// equal steps of time, and when the corner reacts. The question screen
// samples those steps on the UI thread (`sampleShot`).
// The shapes are the design prototype's generator, with its fixed 390 × 844
// positions made relative to the corner, the tap and the screen.

export interface Point {
  x: number;
  y: number;
}

export interface Box extends Point {
  width: number;
  height: number;
}

export type ShotVerdict = 'goal' | 'miss';
export const GOAL_SHOTS = ['Curler', 'Driven', 'Dipping lob', 'Trivela'] as const;
export const MISS_SHOTS = ['Wide right', 'Wide left', 'Sliced right', 'Dragged left'] as const;
export type GoalShot = (typeof GOAL_SHOTS)[number];
export type MissShot = (typeof MISS_SHOTS)[number];
export type ShotName = GoalShot | MissShot;

/** The ball at equal steps of time over its flight: its centre in the
 *  overlay's coordinates, its size × `sizes.corner.question.ball`, and its
 *  opacity. */
export interface ShotFrames {
  flight: number;
  x: number[];
  y: number[];
  scale: number[];
  opacity: number[];
}

export interface ShotPlan {
  name: ShotName;
  verdict: ShotVerdict;
  frames: ShotFrames;
  /** ms after the strike when the corner reacts: a goal landing, a miss going by. */
  impact: number;
  /** ms after the strike when the ball, its trail and the corner are all still. */
  end: number;
}

export interface ShotInput {
  verdict: ShotVerdict;
  /** Where the ball is struck from (`shotOrigin`). */
  from: Point;
  /** The top-left of the corner's arm × arm box. */
  corner: Point;
  /** The overlay's size: a miss leaves past its edges. */
  bounds: { width: number; height: number };
  seed: number;
  /** The last shot with the same verdict: never played twice running. */
  previous: ShotName | null;
}

/** The shot as the corner and the overlay read it on the UI thread. `frames`
 *  is null when nothing flies (Reduce Motion): the corner only turns. */
export interface LiveShot {
  question: number;
  verdict: ShotVerdict;
  impact: number;
  frames: ShotFrames | null;
}

export interface ShotSample {
  x: number;
  y: number;
  scale: number;
  opacity: number;
}

const { arm, bar, ball } = sizes.corner.question;

/** The ball's centre at rest, from the corner box's top-left: just under the
 *  crossbar, centred in the space left of the post. */
export const CORNER_BALL: Point = { x: (arm - bar) / 2, y: bar + ball / 2 };

type Rng = () => number;
type Segment = readonly [Point, Point, Point, Point];

// Steps per flight: a 300 ms flight at 120 Hz draws 36 frames.
const STEPS = 48;
// Control points stay this far inside the screen's sides and above its
// bottom, and no further than `CEILING` above its top.
const EDGE = 14;
const FLOOR = 44;
const CEILING = 200;
// A miss leaves this far past the screen's side.
const OFF = 120;
// Tries at one miss shape before the next is tried.
const REROLLS = 8;
// A goal leaves the boot fast and slows into the net.
const GOAL_EASE = 2.1;
// The ball fades in over the first 8 % of its flight; a miss fades out over
// the last 12 % of its path.
const FADE_IN = 0.08;
const FADE_OUT = 0.88;
// A miss keeps its pace, a little slower than the average goal.
const MISS_SPEED = 1.15;

interface Path {
  name: ShotName;
  segs: readonly Segment[];
  /** × `durations.shotFlight`. */
  speed: number;

const add = (a: Point, b: Point): Point => ({ x: a.x + b.x, y: a.y + b.y });
const sub = (a: Point, b: Point): Point => ({ x: a.x - b.x, y: a.y - b.y });
const mul = (a: Point, k: number): Point => ({ x: a.x * k, y: a.y * k });
const len = (a: Point) => Math.hypot(a.x, a.y);
const norm = (a: Point): Point => mul(a, 1 / (len(a) || 1));
function rot(a: Point, deg: number): Point {
  const q = (deg * Math.PI) / 180;
  return { x: a.x * Math.cos(q) - a.y * Math.sin(q), y: a.x * Math.sin(q) + a.y * Math.cos(q) };
}

function cubic([p0, p1, p2, p3]: Segment, t: number): Point {
  const u = 1 - t;
  const w = [u * u * u, 3 * u * u * t, 3 * u * t * t, t * t * t] as const;
  return {
    x: w[0] * p0.x + w[1] * p1.x + w[2] * p2.x + w[3] * p3.x,
    y: w[0] * p0.y + w[1] * p1.y + w[2] * p2.y + w[3] * p3.y,
  };
}

interface Trace {
  pts: Point[];
  cum: number[];
  total: number;
}

// The path as points close together, with the distance along it to each.
function trace(segs: readonly Segment[]): Trace {
  const pts: Point[] = [];
  segs.forEach((seg, si) => {
    for (let j = si === 0 ? 0 : 1; j <= 48; j++) pts.push(cubic(seg, j / 48));
  });
  const cum = [0];
  for (let i = 1; i < pts.length; i++) {
    const a = pts[i - 1];
    const b = pts[i];
    cum.push((cum[i - 1] ?? 0) + (a && b ? len(sub(b, a)) : 0));
  }
  return { pts, cum, total: cum[cum.length - 1] || 1 };
}

// The point a share `f` of the way along the path.
function along(tr: Trace, f: number): Point {
  const d = f * tr.total;
  let lo = 0;
  let hi = tr.cum.length - 1;
  while (hi - lo > 1) {
    const mid = (lo + hi) >> 1;
    if ((tr.cum[mid] ?? 0) < d) lo = mid;
    else hi = mid;
  }
  const a = tr.pts[lo] ?? { x: 0, y: 0 };
  const b = tr.pts[hi] ?? a;
  const from = tr.cum[lo] ?? 0;
  const span = (tr.cum[hi] ?? from) - from || 1;
  const t = Math.min(1, Math.max(0, (d - from) / span));
  return { x: lerp(a.x, b.x, t), y: lerp(a.y, b.y, t) };
}

function framesOf(path: Path, flight: number): ShotFrames {
  const tr = trace(path.segs);
  const frames: ShotFrames = { flight, x: [], y: [], scale: [], opacity: [] };
  for (let i = 0; i <= STEPS; i++) {
    const tau = i / STEPS;
    const f = path.ease(tau);
    const at = along(tr, f);
    frames.x.push(at.x);
    frames.y.push(at.y);
    frames.scale.push(path.scale(f));
    frames.opacity.push(Math.min(1, tau / FADE_IN) * path.fade(f));
  }
  return frames;
}

function goalPath(input: ShotInput, shape: GoalShape, r: Rng, clamp: (p: Point) => Point): Path {
  const { from, corner } = input;
  const rest = add(corner, CORNER_BALL);
  const side = r() < 0.5 ? -1 : 1;
  const jitter = 0.85 + r() * 0.3;
  const v = sub(rest, from);
  const L = len(v);
  const u = norm(v);
  const n = { x: -u.y, y: u.x };
  // In from the lower left, give or take 15°.
  const open = rot(norm({ x: -0.6, y: 0.8 }), (r() - 0.5) * 30);
  const a = norm(add(mul(u, -(1 - shape.approach)), mul(open, shape.approach)));
  const p1 = clamp(add(add(from, mul(u, 0.35 * L)), mul(n, side * shape.bend1 * L * jitter)));
  const p2 = clamp(add(add(rest, mul(a, 0.3 * L)), mul(n, side * shape.bend2 * L * jitter)));
  p2.y = Math.max(p2.y, rest.y + 8);
  // The last stretch heads for the rest point from p2. Keep it far enough
  // left that the ball clears the post as it rises past the box's bottom.
  const postLeft = corner.x + arm - bar;
  const room = postLeft - ball / 2 - 1 - rest.x;
  p2.x = Math.min(p2.x, rest.x + (room * (p2.y - rest.y)) / (corner.y + arm - rest.y));
  const swell = shape.swell * jitter;
  const { startScale } = spotlightMotion.shot;
  return {
    name: shape.name,
    segs: [[from, p1, p2, rest]],
    speed: shape.speed,
    mark: 1,
    ease: (t) => 1 - Math.pow(1 - t, GOAL_EASE),
    // Lob swell: from the start scale down to the corner's ball, bigger at
    // the top of the arc.
    scale: (f) => startScale - (startScale - 1) * f + swell * Math.sin(Math.PI * f),
    fade: () => 1,
  };
}

function missOf(name: MissShot, segs: readonly Segment[], mark: number, grow: number): Path {
  const { startScale } = spotlightMotion.shot;
  return {
    name,
    segs,
    speed: MISS_SPEED,
    mark,
    ease: (t) => t,
    // It rises to its aim point, then grows as it comes off the screen.
    scale: (f) => {
      const g = f <= mark ? f / mark : 1;
      const h = f <= mark ? 0 : (f - mark) / (1 - mark);
      return startScale - 0.5 * g + 0.6 * Math.sin(Math.PI * g) + grow * Math.pow(h, 1.3);
    },
    fade: (f) => (f > FADE_OUT ? Math.max(0, (1 - f) / (1 - FADE_OUT)) : 1),
  };
}

function missPath(input: ShotInput, shape: MissShape, r: Rng, clamp: (p: Point) => Point): Path {
  const { from } = input;
  const c: MissContext = { from, corner: input.corner, width: input.bounds.width };
  const aim = shape.aim(c, r);
  const exit = shape.exit(c, aim, r);
  const side = r() < 0.5 ? -1 : 1;
  const v = sub(aim, from);
  const L1 = len(v);
  const u = norm(v);
  const n = { x: -u.y, y: u.x };
  const a = rot(norm(shape.approach), (r() - 0.5) * 16);
  const p1 = clamp(add(add(from, mul(u, 0.35 * L1)), mul(n, side * 0.1 * L1)));
  const p2 = add(aim, mul(a, 0.3 * L1));
  const dIn = norm(sub(aim, p2));
  const L2 = len(sub(exit, aim));
  const q1 = add(aim, mul(dIn, 0.35 * L2));
  const q2 = add(exit, mul(norm(sub(exit, q1)), -0.25 * L2));
  const rise: Segment = [from, p1, p2, aim];
  const off: Segment = [aim, q1, q2, exit];
  const mark = trace([rise]).total / trace([rise, off]).total;
  return missOf(shape.name, [rise, off], mark, shape.grow);
}

// The last resort, never needed on a real layout: level across the screen
// from the tap, which sits well below the corner.
function flatMiss(input: ShotInput, toRight: boolean): Path {
  const { from } = input;
  const exit = { x: toRight ? input.bounds.width + OFF : -OFF, y: from.y };
  const third = mul(sub(exit, from), 1 / 3);
  const seg: Segment = [from, add(from, third), add(from, mul(third, 2)), exit];
  return missOf(toRight ? 'Sliced right' : 'Dragged left', [seg], 0.5, 0.8);
}

// How close the ball's edge comes to the corner's box, as drawn.
function closest(frames: ShotFrames, corner: Point): number {
  let best = Infinity;
  for (let ms = 0; ms < frames.flight; ms += 1) {
    const at = sampleShot(frames, ms);
    if (!at) continue;
    const dx = Math.max(corner.x - at.x, 0, at.x - (corner.x + arm));
    const dy = Math.max(corner.y - at.y, 0, at.y - (corner.y + arm));
    best = Math.min(best, Math.hypot(dx, dy) - (ball / 2) * at.scale);
  }
  return best;
}

function item<T>(list: readonly T[], k: number): T {
  const found = list[k];
  if (found === undefined) throw new Error(`No shot at ${k}`);
  return found;
}

// A shape at random, never the previous one.
function pick(names: readonly ShotName[], previous: ShotName | null, r: Rng): number {
  let k = Math.floor(r() * names.length);
  if (names[k] === previous) k = (k + 1 + Math.floor(r() * (names.length - 1))) % names.length;
  return k;
}

const TRAIL_END = Math.max(...spotlightMotion.shot.trail.map((ghost) => ghost.delay));

export function planShot(input: ShotInput): ShotPlan {
  const r = createRng(input.seed);
  const { width, height } = input.bounds;
  const clamp = (p: Point): Point => ({
    x: Math.min(width - EDGE, Math.max(EDGE, p.x)),
    y: Math.min(height - FLOOR, Math.max(-CEILING, p.y)),
  });

  if (input.verdict === 'goal') {
    const shape = item(GOALS, pick(GOAL_SHOTS, input.previous, r));
    const path = goalPath(input, shape, r, clamp);
    const flight = Math.round(durations.shotFlight * path.speed);
    const after = Math.max(durations.shotSettle, durations.cornerShake, durations.ring, TRAIL_END);
    return { name: path.name, verdict: 'goal', frames: framesOf(path, flight), impact: flight, end: flight + after };
  }

  const flight = Math.round(durations.shotFlight * MISS_SPEED);
  const miss = (path: Path): ShotPlan => ({
    name: path.name,
    verdict: 'miss',
    frames: framesOf(path, flight),
    impact: Math.round(flight * path.mark),
    end: flight + TRAIL_END,
  });
  // The shape picked first, then the others; each re-rolled until it stays
  // clear of the corner.
  const first = pick(MISS_SHOTS, input.previous, r);
  const order = [first, ...MISSES.map((_, k) => k).filter((k) => k !== first && MISS_SHOTS[k] !== input.previous)];
  for (const k of order) {
    for (let i = 0; i < REROLLS; i++) {
      const plan = miss(missPath(input, item(MISSES, k), r, clamp));
      if (closest(plan.frames, input.corner) >= spotlightMotion.shot.clearance) return plan;
    }
  }
  const away = input.corner.x + arm / 2 < width / 2;
  const right = away ? input.previous !== 'Sliced right' : input.previous === 'Dragged left';
  return miss(flatMiss(input, right));
}

function mix(values: number[], i: number, t: number): number {
  'worklet';
  const a = values[i] ?? 0;
  const b = values[i + 1] ?? a;
  return a + (b - a) * t;
}

/** The ball `ms` after the strike, blended between the planned steps; null
 *  before the strike and from the end of the flight on. A worklet: the
 *  overlay calls it on the UI thread every frame. */
export function sampleShot(frames: ShotFrames, ms: number): ShotSample | null {
  'worklet';
  const steps = frames.x.length - 1;
  if (steps < 1 || !(ms >= 0) || ms >= frames.flight) return null;
  const at = (ms / frames.flight) * steps;
  const i = Math.floor(at);
  const t = at - i;
  return {
    x: mix(frames.x, i, t),
    y: mix(frames.y, i, t),
    scale: mix(frames.scale, i, t),
    opacity: mix(frames.opacity, i, t),
  };
}

/** Where the ball is struck from: the tap, when the press carried one on the
 *  tile, else the tile's centre (a screen reader's activation carries none). */
export function shotOrigin(tap: Point | null, tile: Box | null): Point | null {
  if (!tile) return tap;
  const onTile =
    tap !== null &&
    tap.x >= tile.x &&
    tap.x <= tile.x + tile.width &&
    tap.y >= tile.y &&
    tap.y <= tile.y + tile.height;
  return onTile ? tap : { x: tile.x + tile.width / 2, y: tile.y + tile.height / 2 };
}

import { useEffect, useRef } from 'react';
import type { View } from 'react-native';
import {
  Easing,
  useReducedMotion,
  useSharedValue,
  withDelay,
  withSequence,
  withTiming,
} from 'react-native-reanimated';
import {
  planShot,
  shotOrigin,
  type Box,
  type LiveShot,
  type Point,
  type ShotName,
  type ShotVerdict,
} from '@/lib/shot';
import { durations, spotlightCues } from '@/theme/tokens';

/** Where a tile was tapped: the press's window coordinates (0, 0 when it
 *  carried none, as a screen reader's activation may) and the tile itself. */
export interface ShotTap {
  pageX: number;
  pageY: number;
  tile: View | null;
}

function measure(view: View | null): Promise<Box | null> {
  return new Promise((resolve) => {
    if (!view) {
      resolve(null);
      return;
    }
    view.measureInWindow((x, y, width, height) => resolve({ x, y, width, height }));
  });
}

// The shot when a question is decided (#129, spec §6). It owns what the
// corner and the overlay read on the UI thread: the shot (`live`) and its
// clock, the ms since the ball left the tile.
// - `fire` waits for the slot to fill (`spotlightCues.slot`), since a long
//   answer can reflow the chips and move the corner. Then it measures the
//   overlay, the corner and the tile, plans the shot and strikes it at
//   `spotlightCues.ball`.
// - With Reduce Motion, or if something can't be measured, nothing flies:
//   the corner just turns.
// - A shot belongs to its question. The next question ignores it, and drops
//   one still waiting to be struck.
export function useShot(question: number) {
  const reduced = useReducedMotion();
  const live = useSharedValue<LiveShot | null>(null);
  const clock = useSharedValue(-1);
  const overlayRef = useRef<View>(null);
  const cornerRef = useRef<View>(null);
  const previous = useRef<Record<ShotVerdict, ShotName | null>>({ goal: null, miss: null });
  const timer = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
  const onScreen = useRef<number | null>(question);

  useEffect(() => {
    onScreen.current = question;
    return () => {
      onScreen.current = null;
      clearTimeout(timer.current);
    };
  }, [question]);

  const strike = (shot: LiveShot, delay: number, end: number) => {
    clock.set(-1);
    live.set(shot);
    clock.set(
      withDelay(
        delay,
        withSequence(
          withTiming(0, { duration: 0 }),
          withTiming(end, { duration: end, easing: Easing.linear }),
        ),
      ),
    );
  };

  const fire = (verdict: ShotVerdict, tap: ShotTap) => {
    const firedAt = Date.now();
    const still = (delay: number) =>
      strike({ question, verdict, impact: 0, frames: null }, delay, durations.transition);
    if (reduced) {
      still(0);
      return;
    }
    clearTimeout(timer.current);
    timer.current = setTimeout(() => {
      void Promise.all([
        measure(overlayRef.current),
        measure(cornerRef.current),
        measure(tap.tile),
      ]).then(([overlay, corner, tile]) => {
        if (onScreen.current !== question) return;
        const wait = Math.max(0, spotlightCues.ball - (Date.now() - firedAt));
        if (!overlay || !corner) {
          still(wait);
          return;
        }
        const local = (p: Point): Point => ({ x: p.x - overlay.x, y: p.y - overlay.y });
        const from = shotOrigin(
          tap.pageX || tap.pageY ? local({ x: tap.pageX, y: tap.pageY }) : null,
          tile ? { ...tile, ...local(tile) } : null,
        );
        if (!from) {
          still(wait);
          return;
        }
        const plan = planShot({
          verdict,
          from,
          corner: local(corner),
          bounds: overlay,
          seed: firedAt,
          previous: previous.current[verdict],
        });
        previous.current[verdict] = plan.name;
        strike({ question, verdict, impact: plan.impact, frames: plan.frames }, wait, plan.end);
      });
    }, spotlightCues.slot);
  };

  return { live, clock, overlayRef, cornerRef, fire };
}

// One ball: the shot itself, or one of its trail, `layer.delay` ms behind,
// fainter and smaller. Drawn large and scaled down, so it stays sharp.
function ShotBall({ layer, question, live, clock, goal, miss }: ShotBallProps) {
  const style = useAnimatedStyle(() => {
    const shot = live.value;
    const at =
      shot && shot.frames && shot.question === question
        ? sampleShot(shot.frames, clock.value - layer.delay)
        : null;
    return {
      backgroundColor: shot?.verdict === 'miss' ? miss : goal,
      opacity: at ? at.opacity * layer.opacity : 0,
      transform: [
        { translateX: at ? at.x : 0 },
        { translateY: at ? at.y : 0 },

import type { Ref } from 'react';
import { StyleSheet, View } from 'react-native';
import Animated, {
  Easing,
  Extrapolation,
  interpolate,
  interpolateColor,
  useAnimatedStyle,
  useReducedMotion,
  type SharedValue,
} from 'react-native-reanimated';
import { CORNER_BALL, type LiveShot } from '@/lib/shot';
import {
  borderWidths,
  celebrationEasingCurves,
  durations,
  sizes,
  spotlightMotion,
} from '@/theme/tokens';
import { useThemeColors } from '@/theme/useTheme';

const { arm, bar, ball } = sizes.corner.question;
const { cornerKick, ring, shot } = spotlightMotion;
// The kick as keyframes over its progress, at rest at both ends; y moves
// opposite to x.
const KICK_AT = [0, ...cornerKick.map((swing) => swing.at), 1];
const KICK_X = [0, ...cornerKick.map((swing) => swing.x), 0];
const KICK_SCALE = [1, ...cornerKick.map((swing) => swing.scale), 1];
const kickEasing = Easing.out(Easing.quad);
const ringEasing = Easing.bezierFn(...celebrationEasingCurves.rise);

interface EscuadraCornerProps {
  /** The question on screen: a shot for another question (the one sliding
   *  out) leaves this corner as it is. */
  question: number;
  /** The shot and its clock, the ms since the ball left the tile (`useShot`). */
  live: SharedValue<LiveShot | null>;
  clock: SharedValue<number>;
  /** The corner's box, which never moves: the shot measures it to aim. */
  ref?: Ref<View>;
}

// The escuadra corner at the shirt number's top right (spec §2.1, §6, #129).
// It reacts on the shot's clock, so it turns on the frame the ball arrives.
// - Asking: accent.
// - A goal: as the ball lands, the corner turns green, the frame kicks, a
//   ring flashes out and the ball settles under the crossbar, where it stays.
// - A miss: it turns red as the ball goes by. The frame never moves.
// Under Reduce Motion nothing flies (`useShot`). The colour turns, the ball
// fades in on a goal, and nothing moves.
export function EscuadraCorner({ question, live, clock, ref }: EscuadraCornerProps) {
  const colors = useThemeColors();
  const reduced = useReducedMotion();
  const asking = colors.accentLine;
  const goal = colors.success;
  const miss = colors.error;
  const turn = durations.transition;
  const kick = durations.cornerShake;
  const flash = durations.ring;
  const settle = durations.shotSettle;

  const bars = useAnimatedStyle(() => {
    const s = live.value;
    const since = s && s.question === question ? clock.value - s.impact : -1;
    const turned = since < 0 ? 0 : Math.min(1, since / turn);
    return {
      backgroundColor: interpolateColor(turned, [0, 1], [asking, s?.verdict === 'miss' ? miss : goal]),
    };
  });

  const frame = useAnimatedStyle(() => {
    const s = live.value;
    const kicking = s && s.question === question && s.verdict === 'goal' && !reduced;
    const since = kicking ? clock.value - s.impact : -1;
    const p = since < 0 || since >= kick ? 0 : kickEasing(since / kick);
    const x = interpolate(p, KICK_AT, KICK_X);
    return {
      transform: [
        { translateX: x },
        { translateY: -x },
        { scale: interpolate(p, KICK_AT, KICK_SCALE) },
      ],
    };
  });

  const flashStyle = useAnimatedStyle(() => {
    const s = live.value;
    const flashing = s && s.question === question && s.verdict === 'goal' && !reduced;
    const since = flashing ? clock.value - s.impact : -1;
    if (since < 0 || since >= flash) return { opacity: 0, transform: [{ scale: ring.from }] };
    const p = ringEasing(since / flash);
    return {
      opacity: interpolate(p, [0, 1], [ring.opacity, 0]),
      transform: [{ scale: interpolate(p, [0, 1], [ring.from, ring.to]) }],
    };
  });

  const ballStyle = useAnimatedStyle(() => {
    const s = live.value;
    const since = s && s.question === question && s.verdict === 'goal' ? clock.value - s.impact : -1;
    if (since < 0) return { opacity: 0, transform: [{ scale: 1 }] };
    if (reduced) return { opacity: Math.min(1, since / turn), transform: [{ scale: 1 }] };
    return {
      opacity: 1,
      transform: [
        {
          scale: interpolate(
            since,
            [0, settle / 2, settle],
            [1, shot.settleScale, 1],
            Extrapolation.CLAMP,
          ),
        },
      ],
    };
  });

  const styles = StyleSheet.create({
    corner: { position: 'absolute', top: 0, right: 0, width: arm, height: arm },
    // It kicks from the corner's top right, where the bars meet.
    frame: { transformOrigin: 'top right' },
    crossbar: { position: 'absolute', top: 0, right: 0, width: arm, height: bar },
    post: { position: 'absolute', top: 0, right: 0, width: bar, height: arm },
    ring: {
      position: 'absolute',
      left: CORNER_BALL.x - arm / 2,
      top: CORNER_BALL.y - arm / 2,
      width: arm,
      height: arm,
      borderRadius: arm / 2,
      borderWidth: borderWidths.emphasis,
      borderColor: goal,
    },
    ball: {
      position: 'absolute',
      left: CORNER_BALL.x - ball / 2,
      top: CORNER_BALL.y - ball / 2,
      width: ball,
      height: ball,
      borderRadius: ball / 2,
      backgroundColor: goal,
    },
  });

  return (
    <View
      ref={ref}
      pointerEvents="none"
      style={styles.corner}
      accessibilityElementsHidden
      importantForAccessibility="no-hide-descendants"
    >
      <Animated.View style={[StyleSheet.absoluteFill, styles.frame, frame]}>
        <Animated.View style={[styles.crossbar, bars]} />
        <Animated.View style={[styles.post, bars]} />
      </Animated.View>
      <Animated.View style={[styles.ring, flashStyle]} />
      <Animated.View style={[styles.ball, ballStyle]} />
    </View>
  );
}
